      {isEqModalOpen && createPortal(
        <div className="fixed inset-0 bg-black/60 backdrop-blur-md flex items-center justify-center p-4 z-50 animate-fade-in" id="eq-modal">
          <div className="bg-orange-50/95 backdrop-blur-sm rounded-[2.5rem] border-4 border-white shadow-[0_25px_50px_-12px_rgba(234,88,12,0.25)] max-w-3xl w-full flex flex-col max-h-[90vh] animate-slide-up relative overflow-hidden font-sans text-gray-800">
            {/* Top Banner Accent */}
            <div className="absolute top-0 left-0 right-0 h-2 bg-gradient-to-r from-orange-400 via-amber-400 to-orange-500 z-10" />

            {/* Header */}
            <div className="bg-gradient-to-br from-orange-500 to-amber-500 text-white p-5 sm:p-6 relative flex items-center gap-4 border-b-4 border-white/50 shrink-0">
              <button 
                type="button"
                onClick={() => setIsEqModalOpen(false)} 
                className="absolute top-4 right-4 text-white/80 hover:text-white bg-white/10 hover:bg-white/20 p-2 rounded-full transition-all hover:rotate-90 active:scale-90"
              >
                <X className="w-5 h-5" />
              </button>
              <div className="w-10 h-10 rounded-xl overflow-hidden border border-white/50 shadow-md bg-white shrink-0 flex items-center justify-center text-xl">
                📦
              </div>
              <div>
                <h3 className="font-heading font-black text-sm sm:text-base tracking-wide drop-shadow-sm">
                  {editingEqId ? '✏️ แก้ไขข้อมูลพัสดุครุภัณฑ์' : 'ขึ้นทะเบียนอุปกรณ์ชิ้นใหม่'}
                </h3>
                <p className="text-orange-50 text-[10px] sm:text-xs mt-0.5 font-medium">เพิ่มหรือปรับปรุงข้อมูลพัสดุในคลังของ อบต.ม่อนจอง</p>
              </div>
            </div>

            <form onSubmit={handleSaveEquipmentSubmit} className="p-6 space-y-4 overflow-y-auto custom-scrollbar">
              <div>
                <label className="block text-xs font-bold text-gray-700 mb-1 font-heading">ชื่อครุภัณฑ์ / รายการพัสดุ <span className="text-red-500">*</span></label>
                <input
                  type="text"
                  required
                  value={eqName}
                  onChange={(e) => setEqName(e.target.value)}
                  placeholder=""
                  className="w-full p-2.5 bg-orange-50/60 focus:bg-orange-50/90 backdrop-blur-xs border border-gray-200 rounded-xl text-xs outline-none focus:border-orange-500 focus:ring-2 focus:ring-orange-200 transition-all"
                />
              </div>

              <div className="grid grid-cols-2 gap-3">
                <div>
                  <label className="block text-xs font-bold text-gray-700 mb-1 font-heading">รหัส/หมายเลขครุภัณฑ์</label>
                  <input
                    type="text"
                    
                    value={eqCode}
                    onChange={(e) => setEqCode(e.target.value)}
                    placeholder=""
                    className="w-full p-2.5 bg-orange-50/60 focus:bg-orange-50/90 backdrop-blur-xs border border-gray-200 rounded-xl text-xs outline-none focus:border-orange-500 focus:ring-2 focus:ring-orange-200 transition-all"
                  />
                </div>

                <div>
                  <label className="block text-xs font-bold text-gray-700 mb-1 font-heading">หมวดหมู่ประเภท <span className="text-red-500">*</span></label>
                  <input
                    type="text"
                    required
                    value={eqType}
                    onChange={(e) => setEqType(e.target.value)}
                    placeholder=""
                    className="w-full p-2.5 bg-orange-50/60 focus:bg-orange-50/90 backdrop-blur-xs border border-gray-200 rounded-xl text-xs outline-none focus:border-orange-500 focus:ring-2 focus:ring-orange-200 transition-all"
                  />
                </div>
              </div>

              <div className="grid grid-cols-2 gap-3">
                <div>
                  <label className="block text-xs font-bold text-gray-700 mb-1 font-heading">สถานะพัสดุ</label>
                  <select
                    value={eqStatus}
                    onChange={(e: any) => setEqStatus(e.target.value)}
                    className="w-full p-2.5 bg-orange-50/60 focus:bg-orange-50/90 backdrop-blur-xs border border-gray-200 rounded-xl text-xs outline-none focus:border-orange-500 focus:ring-2 focus:ring-orange-200 font-heading transition-all"
                  >
                    <option value="ปกติ">ปกติ (ใช้งานได้)</option>
                    <option value="ชำรุด">ชำรุด</option>
                    <option value="เสีย">เสีย</option>
                    <option value="จำหน่าย">จำหน่าย/ตัดสต็อก</option>
                  </select>
                </div>

                <div>
                  <label className="block text-xs font-bold text-gray-700 mb-1 font-heading">จำนวนคงคลังนำขึ้นทะเบียน <span className="text-red-500">*</span></label>
                  <input
                    type="text"
                    required
                    value={eqQty}
                    onChange={(e) => setEqQty(e.target.value.replace(/[^0-9]/g, ''))}
                    className="w-full p-2.5 bg-orange-50/60 focus:bg-orange-50/90 backdrop-blur-xs border border-gray-200 rounded-xl text-xs outline-none focus:border-orange-500 focus:ring-2 focus:ring-orange-200 font-mono transition-all"
                  />
                </div>
              </div>

              <div>
                <label className="block text-xs font-bold text-gray-700 mb-1 font-heading">สถานที่จัดเก็บ <span className="text-red-500">*</span></label>
                <input
                  type="text"
                  required
                  value={eqLocation}
                  onChange={(e) => setEqLocation(e.target.value)}
                  placeholder=""
                  className="w-full p-2.5 bg-orange-50/60 focus:bg-orange-50/90 backdrop-blur-xs border border-gray-200 rounded-xl text-xs outline-none focus:border-orange-500 focus:ring-2 focus:ring-orange-200 transition-all"
                />
              </div>

              <div>
                <label className="block text-xs font-bold text-gray-700 mb-1 font-heading">หมายเหตุประกอบเพิ่มเติม</label>
                <textarea
                  value={eqNote}
                  onChange={(e) => setEqNote(e.target.value)}
                  placeholder=""
                  rows={2}
                  className="w-full p-2.5 bg-orange-50/60 focus:bg-orange-50/90 backdrop-blur-xs border border-gray-200 rounded-xl text-xs outline-none focus:border-orange-500 focus:ring-2 focus:ring-orange-200 font-sans transition-all"
                />
              </div>

              <div className="flex space-x-3 pt-2">
                <motion.button 
                  whileHover={{ scale: 1.02 }}
                  whileTap={{ scale: 0.98 }}
                  type="button" 
                  onClick={() => setIsEqModalOpen(false)}
                  className="flex-1 py-3 rounded-2xl border-2 border-gray-300 text-gray-600 bg-white text-xs font-black font-heading hover:bg-gray-50 active:translate-y-[2px] shadow-sm transition-all animate-none"
                >
                  ยกเลิก ✖
                </motion.button>
                <motion.button 
                  whileHover={{ scale: 1.02 }}
                  whileTap={{ scale: 0.98 }}
                  type="submit"
                  className="flex-[2] py-3 rounded-2xl bg-gradient-to-r from-orange-500 to-amber-500 text-white text-xs font-black font-heading flex items-center justify-center space-x-2 shadow-[0_4px_0_0_#ea580c] active:translate-y-[2px] active:shadow-[0_2px_0_0_#ea580c] transition-all"
                >
                  <span>{editingEqId ? 'บันทึกแก้ไขพัสดุ 💾' : 'บันทึกขึ้นทะเบียนพัสดุ 💾'}</span>
                </motion.button>
              </div>
            </form>
          </div>
        </div>,
        document.body
      )}

      {/* --- MODAL 2: ADD/EDIT BORROW TRANSACTION --- */}
      {isBorrowModalOpen && createPortal(
        <div className="fixed inset-0 bg-black/60 backdrop-blur-md flex items-center justify-center p-4 z-50 animate-fade-in" id="borrow-modal">
          <div className="bg-orange-50/95 backdrop-blur-sm rounded-[2.5rem] border-4 border-white shadow-[0_25px_50px_-12px_rgba(234,88,12,0.25)] max-w-3xl w-full flex flex-col max-h-[90vh] animate-slide-up relative overflow-hidden font-sans text-gray-800">
            {/* Top Banner Accent */}
            <div className="absolute top-0 left-0 right-0 h-2 bg-gradient-to-r from-orange-400 via-amber-400 to-orange-500 z-10" />

            {/* Header */}
            <div className="bg-gradient-to-br from-orange-500 to-amber-500 text-white p-5 sm:p-6 relative flex items-center gap-4 border-b-4 border-white/50 shrink-0">
              <button 
                type="button"
                onClick={() => setIsBorrowModalOpen(false)} 
                className="absolute top-4 right-4 text-white/80 hover:text-white bg-white/10 hover:bg-white/20 p-2 rounded-full transition-all hover:rotate-90 active:scale-90"
              >
                <X className="w-5 h-5" />
              </button>
              <div className="w-10 h-10 rounded-xl overflow-hidden border border-white/50 shadow-md bg-white shrink-0 flex items-center justify-center text-xl">
                🤝
              </div>
              <div>
                <h3 className="font-heading font-black text-sm sm:text-base tracking-wide drop-shadow-sm flex items-center gap-2">
                  <span>{editingBorrowId ? '✏️ ปรับปรุงสลิปยืมอุปกรณ์' : 'ยืมและส่งมอบอุปกรณ์พัสดุ'}</span>
                </h3>
                <p className="text-orange-50 text-[10px] sm:text-xs mt-0.5 font-medium">บันทึกสลิปใบยืมครุภัณฑ์พร้อมหักจำนวนสต็อก</p>
              </div>
            </div>

            <form onSubmit={handleBorrowSubmit} className="p-6 space-y-4 overflow-y-auto custom-scrollbar">
              <div className="grid grid-cols-2 gap-3">
                <div>
                  <label className="block text-xs font-bold text-gray-700 mb-1 font-heading">วันที่ยืม <span className="text-red-500">*</span></label>
                  <input
                    type="date"
                    required
                    value={bDate}
                    onChange={(e) => setBDate(e.target.value)}
                    className="w-full p-2.5 bg-orange-50/60 focus:bg-orange-50/90 backdrop-blur-xs border border-gray-200 rounded-xl text-xs outline-none focus:border-orange-500 focus:ring-2 focus:ring-orange-200 font-mono transition-all"
                  />
                </div>

                <div>
                  <label className="block text-xs font-bold text-gray-700 mb-1 font-heading">กำหนดส่งคืน <span className="text-red-500">*</span></label>
                  <input
                    type="date"
                    required
                    value={bReturnDate}
                    onChange={(e) => setBReturnDate(e.target.value)}
                    className="w-full p-2.5 bg-orange-50/60 focus:bg-orange-50/90 backdrop-blur-xs border border-gray-200 rounded-xl text-xs outline-none focus:border-orange-500 focus:ring-2 focus:ring-orange-200 font-mono transition-all"
                  />
                </div>
              </div>

              <div>
                <label className="block text-xs font-bold text-gray-700 mb-1 font-heading flex justify-between items-center">
                  <span>เลือกอุปกรณ์ที่ต้องการยืม <span className="text-red-500">*</span></span>
                  {!editingBorrowId && (
                    <span className="text-[11px] text-orange-600 font-normal">
                      เลือกแล้ว {bItemIds.length} รายการ
                    </span>
                  )}
                </label>
                
                {editingBorrowId ? (
                  <div className="bg-orange-50 border border-orange-200 rounded-xl p-3 flex flex-col sm:flex-row sm:items-center justify-between gap-3">
                    <div>
                      <div className="font-bold text-xs text-orange-950">
                        {equipment.find(item => item.id === bItemIds[0])?.name || 'ครุภัณฑ์'}
                      </div>
                      <div className="text-[11px] text-gray-500 font-mono">
                        ({equipment.find(item => item.id === bItemIds[0])?.code})
                      </div>
                    </div>
                    <div className="flex items-center gap-2">
                      <span className="text-xs font-bold text-gray-700">จำนวนยืม:</span>
                      <input
                        type="number"
                        min={1}
                        value={bItemQuantities[bItemIds[0]] || 1}
                        onChange={(e) => {
                          const val = parseInt(e.target.value, 10);
                          if (!isNaN(val) && val > 0) {
                            setBItemQuantities(prev => ({ ...prev, [bItemIds[0]]: val }));
                          }
                        }}
                        className="w-16 p-1 text-center bg-white border border-gray-300 rounded-lg text-xs font-mono font-bold outline-none focus:ring-2 focus:ring-orange-300"
                      />
                      <span className="text-xs text-gray-600">ชิ้น</span>
                    </div>
                  </div>
                ) : (
                  <div className="border border-orange-200 bg-orange-50/40 rounded-xl p-3 max-h-56 overflow-y-auto space-y-2 custom-scrollbar">
                    {equipment.map(item => {
                      const isSelected = bItemIds.includes(item.id);
                      const { total, borrowed, available } = getStockInfo(item);
                      const isDisabled = available <= 0 || item.status !== 'ปกติ';
                      const currentQty = bItemQuantities[item.id] || 1;

                      return (
                        <div
                          key={item.id}
                          className={`p-2.5 rounded-xl text-xs font-heading border transition-all ${
                            isDisabled 
                              ? 'bg-gray-100/70 border-gray-200 text-gray-400 cursor-not-allowed' 
                              : isSelected
                              ? 'bg-orange-100/90 border-orange-400 text-orange-950 shadow-xs'
                              : 'bg-white border-gray-200 hover:border-orange-300 text-gray-700'
                          }`}
                        >
                          <div className="flex items-center justify-between gap-2">
                            <label className="flex items-center gap-2.5 cursor-pointer flex-grow select-none">
                              <input
                                type="checkbox"
                                disabled={isDisabled}
                                checked={isSelected}
                                onChange={(e) => {
                                  if (e.target.checked) {
                                    setBItemIds(prev => [...prev, item.id]);
                                    setBItemQuantities(prev => ({ ...prev, [item.id]: 1 }));
                                  } else {
                                    setBItemIds(prev => prev.filter(id => id !== item.id));
                                    setBItemQuantities(prev => {
                                      const next = { ...prev };
                                      delete next[item.id];
                                      return next;
                                    });
                                  }
                                }}
                                className="w-4 h-4 text-orange-500 accent-orange-500 rounded focus:ring-orange-400 cursor-pointer"
                              />
                              <div>
                                <span className="font-bold">{item.name}</span>
                                <span className="text-gray-500 text-[11px] ml-1.5 font-mono">({item.code})</span>
                              </div>
                            </label>

                            <div className="text-[11px] flex items-center gap-1.5 shrink-0">
                              {isDisabled ? (
                                <span className="text-red-500 font-medium">
                                  {item.status !== 'ปกติ' ? `[${item.status}]` : `ถูกยืมหมดแล้ว (${borrowed}/${total} ชิ้น)`}
                                </span>
                              ) : (
                                <span className="bg-emerald-100 text-emerald-800 px-2 py-0.5 rounded-full font-bold">
                                  คงเหลือ {available} ชิ้น {borrowed > 0 ? `(ยืมอยู่ ${borrowed})` : ''}
                                </span>
                              )}
                            </div>
                          </div>

                          {/* แสดงแถบระบุจำนวนเมื่อเลือกรายการนี้ */}
                          {isSelected && !isDisabled && (
                            <div className="mt-2 pt-2 border-t border-orange-200/60 flex items-center justify-between text-xs bg-white/80 p-1.5 rounded-lg">
                              <span className="font-semibold text-gray-700">ระบุจำนวนที่ยืม:</span>
                              <div className="flex items-center gap-1.5">
                                <button
                                  type="button"
                                  onClick={() => {
                                    if (currentQty > 1) {
                                      setBItemQuantities(prev => ({ ...prev, [item.id]: currentQty - 1 }));
                                    }
                                  }}
                                  disabled={currentQty <= 1}
                                  className="w-6 h-6 rounded-md bg-orange-200 hover:bg-orange-300 active:scale-95 text-orange-900 font-bold flex items-center justify-center transition disabled:opacity-30 disabled:pointer-events-none"
                                >
                                  -
                                </button>
                                <input
                                  type="number"
                                  min={1}
                                  max={available}
                                  value={currentQty}
                                  onChange={(e) => {
                                    const val = parseInt(e.target.value, 10);
                                    if (!isNaN(val)) {
                                      const clamped = Math.max(1, Math.min(available, val));
                                      setBItemQuantities(prev => ({ ...prev, [item.id]: clamped }));
                                    }
                                  }}
                                  className="w-12 text-center py-0.5 bg-orange-50 border border-orange-300 rounded-md text-xs font-mono font-bold outline-none focus:ring-2 focus:ring-orange-300"
                                />
                                <button
                                  type="button"
                                  onClick={() => {
                                    if (currentQty < available) {
                                      setBItemQuantities(prev => ({ ...prev, [item.id]: currentQty + 1 }));
                                    }
                                  }}
                                  disabled={currentQty >= available}
                                  className="w-6 h-6 rounded-md bg-orange-200 hover:bg-orange-300 active:scale-95 text-orange-900 font-bold flex items-center justify-center transition disabled:opacity-30 disabled:pointer-events-none"
                                >
                                  +
                                </button>
                                <span className="text-[11px] text-gray-500 font-medium ml-1">
                                  ชิ้น (ยืมได้สูงสุด {available})
                                </span>
                              </div>
                            </div>
                          )}
                        </div>
                      );
                    })}
                  </div>
                )}
              </div>

              <div>
                <label className="block text-xs font-bold text-gray-700 mb-1 font-heading">ชื่อ - นามสกุล ผู้ยืม <span className="text-red-500">*</span></label>
                <input
                  type="text"
                  required
                  value={bName}
                  onChange={(e) => setBName(e.target.value)}
                  placeholder=""
                  className="w-full p-2.5 bg-orange-50/60 focus:bg-orange-50/90 backdrop-blur-xs border border-gray-200 rounded-xl text-xs outline-none focus:border-orange-500 focus:ring-2 focus:ring-orange-200 transition-all"
                />
              </div>

              <div className="grid grid-cols-2 gap-3">
                <div>
                  <label className="block text-xs font-bold text-gray-700 mb-1 font-heading">หน่วยงาน <span className="text-red-500">*</span></label>
                  <input
                    type="text"
                    required
                    value={bPosition}
                    onChange={(e) => setBPosition(e.target.value)}
                    placeholder=""
                    className="w-full p-2.5 bg-orange-50/60 focus:bg-orange-50/90 backdrop-blur-xs border border-gray-200 rounded-xl text-xs outline-none focus:border-orange-500 focus:ring-2 focus:ring-orange-200 transition-all"
                  />
                </div>

                <div>
                  <label className="block text-xs font-bold text-gray-700 mb-1 font-heading">เบอร์โทรศัพท์ผู้ยืม <span className="text-red-500">*</span></label>
                  <input
                    type="tel"
                    required
                    value={bTel}
                    onChange={(e) => setBTel(e.target.value)}
                    placeholder=""
                    maxLength={10}
                    className="w-full p-2.5 bg-orange-50/60 focus:bg-orange-50/90 backdrop-blur-xs border border-gray-200 rounded-xl text-xs outline-none focus:border-orange-500 focus:ring-2 focus:ring-orange-200 font-mono transition-all"
                  />
                </div>
              </div>

              <div className="grid grid-cols-1 sm:grid-cols-2 gap-3">
                <div>
                  <label className="block text-xs font-bold text-gray-700 mb-1 font-heading">เจ้าหน้าที่ผู้อนุมัติ/ผู้รับรองสิทธิ์ <span className="text-red-500">*</span></label>
                  <input
                    type="text"
                    required
                    value={bCertifier}
                    onChange={(e) => setBCertifier(e.target.value)}
                    placeholder=""
                    className="w-full p-2.5 bg-orange-50/60 focus:bg-orange-50/90 backdrop-blur-xs border border-gray-200 rounded-xl text-xs outline-none focus:border-orange-500 focus:ring-2 focus:ring-orange-200 transition-all"
                  />
                </div>

                <div>
                  <label className="block text-xs font-bold text-gray-700 mb-1 font-heading">ตำแหน่งผู้รับรองสิทธิ์</label>
                  <input
                    type="text"
                    value={bCertifierPosition}
                    onChange={(e) => setBCertifierPosition(e.target.value)}
                    placeholder=""
                    className="w-full p-2.5 bg-orange-50/60 focus:bg-orange-50/90 backdrop-blur-xs border border-gray-200 rounded-xl text-xs outline-none focus:border-orange-500 focus:ring-2 focus:ring-orange-200 transition-all"
                  />
                </div>
              </div>

              <div>
                <label className="block text-xs font-bold text-gray-700 mb-1 font-heading">วัตถุประสงค์ / หมายเหตุเพิ่มเติม</label>
                <textarea
                  value={bNote}
                  onChange={(e) => setBNote(e.target.value)}
                  placeholder=""
                  rows={2}
                  className="w-full p-2.5 bg-orange-50/60 focus:bg-orange-50/90 backdrop-blur-xs border border-gray-200 rounded-xl text-xs outline-none focus:border-orange-500 focus:ring-2 focus:ring-orange-200 font-sans transition-all"
                />
              </div>

              <div className="flex space-x-3 pt-2">
                <motion.button 
                  whileHover={{ scale: 1.02 }}
                  whileTap={{ scale: 0.98 }}
                  type="button" 
                  onClick={() => setIsBorrowModalOpen(false)}
                  className="flex-1 py-3 rounded-2xl border-2 border-gray-300 text-gray-600 bg-white text-xs font-black font-heading hover:bg-gray-50 active:translate-y-[2px] shadow-sm transition-all animate-none"
                >
                  ยกเลิก ✖
                </motion.button>
                <motion.button 
                  whileHover={{ scale: 1.02 }}
                  whileTap={{ scale: 0.98 }}
                  type="submit"
                  className="flex-[2] py-3 rounded-2xl bg-gradient-to-r from-orange-500 to-amber-500 text-white text-xs font-black font-heading flex items-center justify-center space-x-2 shadow-[0_4px_0_0_#ea580c] active:translate-y-[2px] active:shadow-[0_2px_0_0_#ea580c] transition-all"
                >
                  <span>{editingBorrowId ? 'บันทึกปรับรายละเอียดสลิป 💾' : 'อนุมัติการยืมและส่งมอบของ 🤝'}</span>
                </motion.button>
              </div>
            </form>
          </div>
        </div>,
        document.body
      )}
